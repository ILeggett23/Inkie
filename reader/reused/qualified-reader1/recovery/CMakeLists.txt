cmake_minimum_required(VERSION 3.20)
project(InkboundRecoveryOwner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_SKIP_RPATH TRUE)
option(OWNER_VM_TESTS "Build inert VM-only fault helpers" OFF)
option(OWNER_VM_SHORT_DEADLINE "Accelerate inert VM matrix only" OFF)
find_library(SYSTEMD_LIBRARY NAMES systemd libsystemd.so.0 REQUIRED)
add_compile_options(-Wall -Wextra -Werror -Wno-misleading-indentation)
add_link_options(-Wl,-z,relro,-z,now,-z,noexecstack)
add_executable(inkbound-recovery-owner owner.cpp contract.cpp bus.cpp)
target_link_libraries(inkbound-recovery-owner PRIVATE ${SYSTEMD_LIBRARY})
if(OWNER_VM_TESTS)
 add_executable(inkbound-recovery-owner-vm owner.cpp contract.cpp bus.cpp)
 target_compile_definitions(inkbound-recovery-owner-vm PRIVATE OWNER_VM_TEST=1)
 target_link_libraries(inkbound-recovery-owner-vm PRIVATE ${SYSTEMD_LIBRARY})
 add_executable(recovery-client-vm ../tests/inert_client.cpp ../src/recovery_boundary.cpp contract.cpp bus.cpp)
 target_include_directories(recovery-client-vm PRIVATE . ../src)
 target_compile_definitions(recovery-client-vm PRIVATE OWNER_VM_TEST=1)
 if(OWNER_VM_SHORT_DEADLINE)
  target_compile_definitions(inkbound-recovery-owner-vm PRIVATE OWNER_VM_SHORT_DEADLINE=1)
  target_compile_definitions(recovery-client-vm PRIVATE OWNER_VM_SHORT_DEADLINE=1)
 endif()
 target_link_libraries(recovery-client-vm PRIVATE ${SYSTEMD_LIBRARY})
endif()
add_executable(owner-proof-tests ../tests/proof_tests.cpp contract.cpp)
target_include_directories(owner-proof-tests PRIVATE .)
enable_testing()
add_test(NAME owner-adversarial COMMAND owner-proof-tests)
